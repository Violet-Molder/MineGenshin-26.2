@Override
public boolean hurt(DamageSource source, float amount) { // 重写父类的hurt方法，接收伤害源和伤害数值作为参数
    if (this.isInvulnerableTo(source)) { // 检查当前实体是否对指定伤害源具有免疫能力
        return false; // 如果免疫则直接返回false，表示未受到伤害
    } else if (this.level().isClientSide) { // 检查实体所在的世界是否为客户端侧
        return false; // 客户端不处理实际伤害逻辑，返回false
    } else if (this.isDeadOrDying()) { // 检查实体是否已经死亡或正在死亡过程中
        return false; // 已死亡实体无法再次受伤，返回false
    } else if (source.is(DamageTypeTags.IS_FIRE) && this.hasEffect(MobEffects.FIRE_RESISTANCE)) { // 检查伤害源是否为火焰类型且实体是否拥有火焰抗性效果
        return false; // 火焰抗性免疫火焰伤害，返回false
    } else { // 开始主要伤害处理逻辑
        this.damageContainers.push(new net.neoforged.neoforge.common.damagesource.DamageContainer(source, amount)); // 向伤害容器栈推入一个新的伤害容器对象，包含伤害源和原始伤害值
        if (net.neoforged.neoforged.common.CommonHooks.onEntityIncomingDamage(this, this.damageContainers.peek())) return false; // 触发实体即将受伤的钩子事件，如果事件被取消则返回false
        if (this.isSleeping() && !this.level().isClientSide) { // 检查实体是否处于睡眠状态且不在客户端
            this.stopSleeping(); // 停止睡眠状态
        }

        this.noActionTime = 0; // 重置无行动时间计数器为0
        amount = this.damageContainers.peek().getNewDamage(); // 从伤害容器栈顶取出容器并获取当前修改后的伤害值，更新amount变量
        float f = amount; // 创建原始伤害值的副本，用于后续比较
        boolean flag = false; // 布尔标志，标记伤害是否被完全阻挡
        float f1 = 0.0F; // 浮点数变量，记录被护盾阻挡的伤害量
        net.neoforged.neoforged.common.CommonHooks.onDamageBlock // 调用护盾阻挡钩子方法
        LivingShieldBlockEvent ev; // 声明护盾阻挡事件对象，用于处理护盾相关逻辑
        if (amount > 0.0F && (ev = net.neoforged.neoforged.common.CommonHooks.onDamageBlock(this, this.damageContainers.peek(), this.isDamageSourceBlocked(source))).getBlocked()) { // 检查伤害值大于0且伤害被阻挡时创建护盾事件
            this.damageContainers.peek().setBlockedDamage(ev); // 将护盾事件的结果设置到伤害容器栈顶的容器中
            if(ev.shieldDamage() > 0) { // 检查护盾事件返回的护盾损伤值是否大于0
                this.hurtCurrentlyUsedShield(ev.shieldDamage()); // 对当前使用的护盾物品造成耐久损伤
            }
            f1 = ev.getBlockedDamage(); // 获取护盾阻挡事件中被阻挡的伤害量，赋值给f1变量（翻译：获取阻挡的伤害值）
            amount = ev.getDamageContainer().getNewDamage(); // 从护盾事件的伤害容器中获取新的伤害值，更新amount变量
            if (!source.is(DamageTypeTags.IS_PROJECTILE) && source.getDirectEntity() instanceof LivingEntity livingentity) { // 检查伤害源不是弹射物类型且直接实体为生物类型
                this.blockUsingShield(livingentity); // 生物使用护盾进行格挡的处理逻辑
            }

            flag = amount <= 0; // 检查剩余伤害是否小于等于0，如果是则完全被阻挡
        }

        if (source.is(DamageTypeTags.IS_FREEZING) && this.getType().is(EntityTypeTags.FREEZE_HURTS_EXTRA_TYPES)) { // 检查伤害源是否为冰冻类型且实体类型属于额外受冰冻伤害的类型
            amount *= 5.0F; // 将伤害值乘以5，造成5倍冰冻伤害
        }

        if (source.is(DamageTypeTags.DAMAGES_HELMET) && !this.getItemBySlot(EquipmentSlot.HEAD).isEmpty()) { // 检查伤害源是否影响头盔且头部装备槽不为空
            this.hurtHelmet(source, amount); // 对头盔装备造成伤害
            amount *= 0.75F; // 将伤害值乘以0.75，即头盔减少25%伤害
        }

        this.damageContainers.peek().setNewDamage(amount); // 更新伤害容器栈顶容器中的新伤害值
        this.walkAnimation.setSpeed(1.5F); // 设置行走动画速度为1.5倍，使受伤时移动更明显
        boolean flag1 = true; // 布尔标志，用于控制伤害事件的广播
        if ((float)this.invulnerableTime > 10.0F && !source.is(DamageTypeTags.BYPASSES_COOLDOWN)) { // 检查无敌时间是否超过10刻且伤害源不跳过冷却时间
            if (amount <= this.lastHurt) { // 检查当前伤害是否小于等于上次受到的伤害值
                this.damageContainers.pop(); // 从伤害容器栈中弹出顶部容器
                return false; // 返回false，忽略此次伤害
            }

            this.damageContainers.peek().setReduction(net.neoforged.neoforged.common.damagesource.DamageContainer.Reduction.INVULNERABILITY, this.lastHurt); // 在伤害容器中设置无敌减伤
            this.actuallyHurt(source, amount - this.lastHurt); // 对实体施加实际伤害（当前伤害减去上次伤害）
            this.lastHurt = amount; // 更新上次伤害值为当前伤害值
            flag1 = false; // 设置标志为false，表示不广播伤害事件
        } else { // 正常伤害处理流程
            this.lastHurt = amount; // 更新上次伤害值
            this.invulnerableTime = this.damageContainers.peek().getPostAttackInvulnerabilityTicks(); // 从伤害容器获取攻击后无敌时间并设置
            this.actuallyHurt(source, amount); // 施加实际伤害
            this.hurtDuration = 10; // 设置受伤动画持续时间为10刻
            this.hurtTime = this.hurtDuration; // 设置当前受伤时间为持续时间
        }

        amount = this.damageContainers.peek().getNewDamage(); // 从伤害容器获取最终伤害值，更新amount变量
        Entity entity = source.getEntity(); // 获取伤害源实体
        if (entity != null) { // 检查伤害源实体是否存在
            if (entity instanceof LivingEntity livingentity1 // 检查伤害源是否为生物类型
                && !source.is(DamageTypeTags.NO_ANGER) // 检查伤害源是否不产生愤怒
                && (!source.is(DamageTypes.WIND_CHARGE) || !this.getType().is(EntityTypeTags.NO_ANGER_FROM_WIND_CHARGE))) { // 检查是否不是风弹伤害或实体类型不受风弹愤怒影响
                this.setLastHurtByMob(livingentity1); // 设置最后伤害来源生物为当前伤害源
            }

            if (entity instanceof Player player1) { // 检查伤害源是否为玩家类型
                this.lastHurtByPlayerTime = 100; // 设置最后被玩家伤害时间计数器为100
                this.lastHurtByPlayer = player1; // 设置最后伤害玩家为当前玩家
            } else if (entity instanceof TamableAnimal tamableAnimal && tamableAnimal.isTame()) { // 检查伤害源是否为驯服的可驯化动物
                this.lastHurtByPlayerTime = 100; // 设置最后被玩家伤害时间计数器为100
                if (tamableAnimal.getOwner() instanceof Player player) { // 检查驯养动物的主人是否为玩家
                    this.lastHurtByPlayer = player; // 设置最后伤害玩家为主人的玩家
                } else {
                    this.lastHurtByPlayer = null; // 主人不是玩家则设为null
                }
            }
        }

        if (flag1) { // 检查伤害事件广播标志
            if (flag) { // 检查伤害是否被完全阻挡
                this.level().broadcastEntityEvent(this, (byte)29); // 广播护盾完全阻挡事件（ID 29）
            } else {
                this.level().broadcastDamageEvent(this, source); // 广播一般伤害事件
            }

            if (!source.is(DamageTypeTags.NO_IMPACT) && (!flag || amount > 0.0F)) { // 检查伤害源是否产生冲击效果且伤害未被完全阻挡
                this.markHurt(); // 标记实体受伤，触发光环等视觉效果
            }

            if (!source.is(DamageTypeTags.NO_KNOCKBACK)) { // 检查伤害源是否产生击退效果
                double d0 = 0.0; // X轴击退方向变量
                double d1 = 0.0; // Z轴击退方向变量
                if (source.getDirectEntity() instanceof Projectile projectile) { // 检查直接伤害实体是否为弹射物类型
                    DoubleDoubleImmutablePair doubledoubleimmutablepair = projectile.calculateHorizontalHurtKnockbackDirection(this, source); // 计算弹射物造成的水平击退方向（返回X、Z轴方向对）
                    d0 = -doubledoubleimmutablepair.leftDouble(); // 获取左侧double值（X方向）并取反
                    d1 = -doubledoubleimmutablepair.rightDouble(); // 获取右侧double值（Z方向）并取反
                } else if (source.getSourcePosition() != null) { // 检查伤害源位置是否不为空
                    d0 = source.getSourcePosition().x() - this.getX(); // 计算X轴方向差值
                    d1 = source.getSourcePosition().z() - this.getZ(); // 计算Z轴方向差值
                }

                this.knockback(0.4F, d0, d1); // 应用击退效果，力度0.4，方向为d0和d1
                if (!flag) { // 检查伤害未被完全阻挡
                    this.indicateDamage(d0, d1); // 显示伤害指示效果
                }
            }
        }

        if (this.isDeadOrDying()) { // 检查实体是否已经死亡或正在死亡
            if (!this.checkTotemDeathProtection(source)) { // 检查是否没有不死图腾保护
                if (flag1) { // 检查伤害事件标志
                    this.makeSound(this.getDeathSound()); // 播放实体死亡音效
                }

                this.die(source); // 执行实体死亡逻辑
            }
        } else if (flag1) { // 如果实体未死亡且事件标志为true
            this.playHurtSound(source); // 播放受伤音效
        }

        boolean flag2 = !flag || amount > 0.0F; // 检查伤害未被完全阻挡或剩余伤害大于0，设置最终标志
        if (flag2) { // 如果实际造成了伤害
            this.lastDamageSource = source; // 记录最后伤害源
            this.lastDamageStamp = this.level().getGameTime(); // 记录最后伤害时间戳

            for (MobEffectInstance mobeffectinstance : this.getActiveEffects()) { // 遍历实体的所有激活状态效果
                mobeffectinstance.onMobHurt(this, source, amount); // 触发每个状态效果的受伤回调
            }
        }

        if (this instanceof ServerPlayer) { // 检查实体是否为服务端玩家
            CriteriaTriggers.ENTITY_HURT_PLAYER.trigger((ServerPlayer)this, source, f, amount, flag); // 触发实体伤害玩家的进度条件
            if (f1 > 0.0F && f1 < 3.4028235E37F) { // 检查被阻挡伤害值是否在合理范围内
                ((ServerPlayer)this).awardStat(Stats.DAMAGE_BLOCKED_BY_SHIELD, Math.round(f1 * 10.0F)); // 给玩家增加护盾阻挡伤害统计数据
            }
        }

        if (entity instanceof ServerPlayer) { // 检查伤害源是否为服务端玩家
            CriteriaTriggers.PLAYER_HURT_ENTITY.trigger((ServerPlayer)entity, this, source, f, amount, flag); // 触发玩家伤害实体的进度条件
        }

        this.damageContainers.pop(); // 从伤害容器栈中弹出顶部容器，清理本次伤害处理
        return flag2; // 返回是否实际造成了伤害的布尔值
    }
}
